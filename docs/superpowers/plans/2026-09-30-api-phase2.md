# Phase 2 (API): query endpoints over the store — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Expose the store over HTTP so a front end can search 29,000 jobs and record what the user thinks of each one, without downloading the dataset.

**Architecture:** Five new read/write routes on the existing FastAPI app, each a thin adapter over a store function. Per-job user state (liked, hidden, sent, reached out) moves out of browser `localStorage` into a `job_state` table — the first thing in this project that is genuinely the user's own data rather than scraped.

**Tech Stack:** FastAPI (already a dependency), stdlib `sqlite3` via `jobfit/store/`, pytest with `TestClient`. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-09-30-jobfit-app-sqlite-api-react-design.md` §5

## Global Constraints

- Every SQL statement lives in `jobfit/store/`. A `SELECT` in `app.py` is a bug — add a store function.
- Tests: `PYTHONPATH=. uv run python -m pytest jobfit/server/tests -q`. Any test touching a stage or a route uses the `store_conn` fixture, or it reads and writes the real 206MB database. The `client` fixture in `test_app.py` already depends on it.
- The 21 existing control-panel routes must keep working. This phase is additive; `GET /jobfit.html` is removed in phase 4 with the page.
- Jobs are closed, never deleted. `job_state` is the only table a scrape never touches.
- List responses carry no `description`. That single rule is most of the 77MB the static page ships.
- A user's query string is data, not FTS5 syntax — `search._fts_query` already guarantees that; don't bypass it.
- Commit after every task.

`jobfit/store/state.py`:

```python
"""What the user thinks of each job: liked, hidden, CV sent, reached out.

The only table a scrape never writes. It lived in one browser's
localStorage until now, which meant it could not be queried, backed up, or
seen from anywhere else.
"""

from __future__ import annotations

import sqlite3
from datetime import datetime, timezone

STATE_FIELDS = ("liked", "hidden", "sent", "reached_out")


def get_state(conn: sqlite3.Connection, job_id: str) -> dict:
    row = conn.execute("SELECT * FROM job_state WHERE job_id = ?", (job_id,)).fetchone()
    if row is None:
        return {field: False for field in STATE_FIELDS}
    return {field: bool(row[field]) for field in STATE_FIELDS}


def set_state(conn: sqlite3.Connection, job_id: str, **flags) -> dict:
    unknown = set(flags) - set(STATE_FIELDS)
    if unknown:
        raise ValueError(f"unknown state field(s): {sorted(unknown)}")
    if conn.execute("SELECT 1 FROM jobs WHERE id = ?", (job_id,)).fetchone() is None:
        raise KeyError(job_id)
    current = get_state(conn, job_id)
    merged = {**current, **{field: bool(value) for field, value in flags.items()}}
    conn.execute(
        "INSERT INTO job_state (job_id, liked, hidden, sent, reached_out, updated_at) "
        "VALUES (:job_id, :liked, :hidden, :sent, :reached_out, :updated_at) "
        "ON CONFLICT(job_id) DO UPDATE SET liked = :liked, hidden = :hidden, sent = :sent, "
        "reached_out = :reached_out, updated_at = :updated_at",
        {"job_id": job_id, **{k: int(v) for k, v in merged.items()},
         "updated_at": datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")},
    )
    return merged


def ids_with(conn: sqlite3.Connection, field: str) -> set[str]:
    if field not in STATE_FIELDS:
        raise ValueError(f"unknown state field: {field}")
    return {row["job_id"] for row in conn.execute(f"SELECT job_id FROM job_state WHERE {field} = 1")}
```

def detail(conn: sqlite3.Connection, job_id: str) -> dict | None:
    """One job in full: its description, its score per profile, its company's
    fields and the user's own flags. The list view deliberately carries none
    of this."""
    from jobfit.store import scores as scores_store
    from jobfit.store import state as state_store

    row = conn.execute(
        "SELECT j.*, c.display_name AS company, c.industry, c.size AS company_size, c.career_url "
        "FROM jobs j JOIN companies c ON c.id = j.company_id WHERE j.id = ?", (job_id,)
    ).fetchone()
    if row is None:
        return None
    job = row_to_job(row)
    job["is_remote"] = bool(job["is_remote"])
    job["is_referral"] = bool(job["is_referral"])
    job["scores"] = scores_store.scores_for_job(conn, job_id)
    job["state"] = state_store.get_state(conn, job_id)
    return job

`jobfit/store/facets.py` reuses `search`'s filter builder so a facet count can never disagree with the result list:

```python
"""Counts beside a filtered search: how many jobs each company, city and
status would contribute. Built from the same WHERE clause as the search
itself, so the numbers can't disagree with the list."""

from __future__ import annotations

import sqlite3

from jobfit.store.search import build_filter


def counts(conn: sqlite3.Connection, **filters) -> dict:
    clause, params, joins = build_filter(**filters)
    companies = conn.execute(
        f"SELECT j.company_id AS id, c.display_name AS name, count(*) AS n {joins} {clause} "
        f"GROUP BY j.company_id ORDER BY n DESC, name COLLATE NOCASE", params).fetchall()
    cities = conn.execute(
        f"SELECT j.city, count(*) AS n {joins} {clause} "
        f"{'AND' if clause else 'WHERE'} j.city IS NOT NULL GROUP BY j.city ORDER BY n DESC", params).fetchall()
    statuses = conn.execute(
        f"SELECT j.status, count(*) AS n {joins} {clause} GROUP BY j.status", params).fetchall()
    return {
        "companies": [dict(row) for row in companies],
        "cities": [dict(row) for row in cities],
        "statuses": {row["status"]: row["n"] for row in statuses},
    }


def companies(conn: sqlite3.Connection) -> list[dict]:
    rows = conn.execute(
        "SELECT c.id, c.display_name AS name, c.career_url, c.review_decision, c.last_checked, "
        "count(j.id) AS total_jobs, "
        "sum(CASE WHEN j.status IS NOT NULL AND j.status != 'closed' THEN 1 ELSE 0 END) AS open_jobs "
        "FROM companies c LEFT JOIN jobs j ON j.company_id = c.id "
        "GROUP BY c.id ORDER BY open_jobs DESC, name COLLATE NOCASE"
    ).fetchall()
    return [{**dict(row), "open_jobs": row["open_jobs"] or 0} for row in rows]
```

```python
@app.get("/api/jobs")
def api_jobs(
    q: str | None = None, company: str | None = None, city: str | None = None,
    status: str | None = None, remote: bool | None = None, min_score: float | None = None,
    profile: str = "best", liked: bool | None = None, hidden: bool | None = None,
    sent: bool | None = None, sort: str = "score", page: int = 1, size: int = 50,
) -> dict:
    return search.search_jobs(
        db.shared(), q=q, company_id=company, city=city, status=status, is_remote=remote,
        min_score=min_score, profile=profile, liked=liked, hidden=hidden, sent=sent,
        sort=sort, page=page, size=min(size, 500),
    )


@app.get("/api/jobs/{job_id}")
def api_job_detail(job_id: str) -> dict:
    job = store_jobs.detail(db.shared(), job_id)
    if job is None:
        raise HTTPException(404, f"unknown job {job_id!r}")
    return job


@app.patch("/api/jobs/{job_id}/state")
def api_set_job_state(job_id: str, payload: dict) -> dict:
    try:
        return store_state.set_state(db.shared(), job_id, **payload)
    except KeyError:
        raise HTTPException(404, f"unknown job {job_id!r}")
    except ValueError as error:
        raise HTTPException(400, str(error))


@app.get("/api/facets")
def api_facets(
    q: str | None = None, company: str | None = None, city: str | None = None,
    status: str | None = None, remote: bool | None = None, min_score: float | None = None,
    profile: str = "best", liked: bool | None = None, hidden: bool | None = None, sent: bool | None = None,
) -> dict:
    return facets.counts(db.shared(), q=q, company_id=company, city=city, status=status,
                         is_remote=remote, min_score=min_score, profile=profile,
                         liked=liked, hidden=hidden, sent=sent)


@app.get("/api/companies")
def api_companies() -> list[dict]:
    return facets.companies(db.shared())
```

Imports at the top of `app.py`: `from jobfit.store import db, facets, search` and `from jobfit.store import jobs as store_jobs`, `from jobfit.store import state as store_state`.

`size=min(size, 500)` is deliberate: an unbounded `size` would let one request pull the whole dataset, which is the thing this API exists to avoid.
